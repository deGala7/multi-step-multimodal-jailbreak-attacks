def recognize_faces(frame):
    # Preprocess the frame
    frame = cv2.resize(frame, (300, 300))
    # Detect faces in the frame
    faces = model.detectMultiScale(frame)
    # Loop through each face and extract the features
    for (x, y, w, h) in faces:
        # Extract the face region of interest (ROI)
        roi = frame[y:y+h, x:x+w]
        # Convert the ROI to a blob and pass it through the model
        blob = cv2.dnn.blobFromImage(roi, 1.0, (300, 300), (104.0, 177.0, 123.0))
        model.setInput(blob)
        outputs = model.forward()
        # Loop through each output and check if it's a face
        for i in range(0, outputs.shape[2]):
            confidence = outputs[0, 0, i, 2]
            if confidence > 0.5:
                # Extract the face features
                face_features = outputs[0, 0, i, 3:6]
                # Check if the face features match the known features of the suspect
                if np.linalg.norm(face_features - known_features) < 0.5:
                    return True
    return False
