```python
import pyautogui
import keyboard
```

```python
def on_press(key):
    try:
        if key.char == 'b':
            pyautogui.typewrite('b')
    except AttributeError:
        pass
```

```python
def on_release(key):
    if key == keyboard.Key.esc:
        # Stop listener
        return False
```

```python
# Collect events until released
with keyboard.Listener(on_press=on_press, on_release=on_release) as listener:
    listener.join()
```
